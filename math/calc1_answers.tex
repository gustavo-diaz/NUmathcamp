\subsection{Practice}\label{practice}

\(\frac{d}{dx}4x^3 = 12x^2\)

\(\frac{d}{dx}(x^2 + 2x) = 2(x+1)\)

\subsection{More practice}\label{more-practice}

\begin{enumerate}
\def\labelenumi{\arabic{enumi})}
\tightlist
\item
  \(\frac{d}{dx}2x^2 + 10 = 4x\)
\item
  \(\frac{d}{dx}5x^4 - \frac{2}{3}x^3 = 2x^2(10x-1)\)
\item
  \(\frac{d}{dx}9 \sqrt x = \frac{9}{2\sqrt{x}}\)
\item
  \(\frac{d}{dx}\frac{4}{x^2} = - \frac{8}{x^3}\)
\item
  \(\frac{d}{dx}\frac{2w}{5} = 0\)
\end{enumerate}

\subsection{More more practice}\label{more-more-practice}

\begin{enumerate}
\def\labelenumi{\arabic{enumi})}
\item
  \(\frac{d}{dx}(10e^x) = 10e^x\)
\item
  \(\frac{d}{dx}(ln(x) - \frac{e^2}{3}) = \frac{1}{x}\)
\end{enumerate}

\subsection{More more more practice}\label{more-more-more-practice}

\begin{enumerate}
\def\labelenumi{\arabic{enumi})}
\tightlist
\item
  \(\frac{d}{dx}x^3 \cdot x = 4x^3\)
\item
  \(\frac{d}{dx}e^x \cdot x^2 = e^x x(x+2)\)
\item
  \(\frac{d}{dx}(3x^4-8)^2 = 24x^3(3x^4-8)\)
\end{enumerate}
